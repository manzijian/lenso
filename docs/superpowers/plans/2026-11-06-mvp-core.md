# LENSO MVP 基础功能实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 让 LENSO 完成最小可用闭环：菜单栏触发 → 区域截图 → 保存原图+缩略图 → 写入 SQLite → 图库展示真实截图 → 预览/复制/删除。

**Architecture:** 遵守 LENSO 架构规范分层：Infrastructure（ScreenCaptureService/SQLiteStore/FileStorage）→ Application（CaptureCoordinator）→ Presentation（现有 SwiftUI 视图改接真实数据）。UI 层已有骨架，本计划只加业务层并替换占位数据。

**Tech Stack:** ScreenCaptureKit、SQLite3（系统 libsqlite3，先不引第三方）、CGImage/ImageIO、NSPasteboard。

**范围约束（YAGNI）：** 不做窗口/全屏模式、不做编辑、OCR、翻译、FTS5、相册标签——后续迭代。只做 region 截图闭环。

```swift
import ScreenCaptureKit
import CoreGraphics

enum CapturePermissionStatus { case authorized, denied, notDetermined }

final class CapturePermissionService {
    /// SCGetSharingType 返回当前屏幕录制授权状态
    static func currentStatus() -> CapturePermissionStatus {
        switch CGGetSharingType() {
        case .authorized: .authorized
        case .denied: .denied
        default: .notDetermined
        }
    }

    /// 触发系统授权弹窗（首次调用 SCShareableContent 时触发）
    static func requestAccess() async -> CapturePermissionStatus {
        _ = try? await SCShareableContent.excludingDesktopWindows(false, onScreenWindowsOnly: true)
        return currentStatus()
    }
}
```

### Task 2: 区域截图引擎

**Files:**
- Create: `LENSO/Infrastructure/ScreenCaptureService.swift`

- [ ] **Step 1: 实现（选区 UI 用系统快照遮罩方案：全屏透明窗口 + 鼠标拖拽画框）**

```swift
import ScreenCaptureKit
import CoreGraphics
import AppKit

struct RegionCaptureResult {
    let image: CGImage
    let rect: CGRect        // 逻辑坐标（屏幕坐标系）
    let displayID: CGDirectDisplayID
}

final class ScreenCaptureService {
    /// 用 SCScreenshotManager 截取指定显示器整个画面，再按选区裁剪
    func captureRegion(_ rect: CGRect, displayID: CGDirectDisplayID) async throws -> RegionCaptureResult {
        let content = try await SCShareableContent.excludingDesktopWindows(false, onScreenWindowsOnly: true)
        guard let display = content.displays.first(where: { $0.displayID == displayID }) else {
            throw CaptureError.displayNotFound(displayID)
        }
        let filter = SCContentFilter(desktopIndependentWindow: display.escapedWindows.first ?? SCWindow()) // 占位，实际用 display filter
        // 正确方式：display 过滤器
        let displayFilter = SCContentFilter(display: display, excludingWindows: [])
        let config = SCStreamConfiguration()
        config.width = Int(display.width)
        config.height = Int(display.height)
        config.scalesToFit = false
        config.showsCursor = false
        let fullImage = try await SCScreenshotManager.captureImage(contentFilter: displayFilter, configuration: config)
        // 像素坐标 = 逻辑坐标 × scale
        let scale = display.scale
        let pxRect = CGRect(
            x: (rect.origin.x - display.frame.origin.x) * scale,
            y: (rect.origin.y - display.frame.origin.y) * scale,
            width: rect.width * scale,
            height: rect.height * scale
        )
        guard let cropped = fullImage.cropping(to: pxRect) else {
            throw CaptureError.cropFailed
        }
        return RegionCaptureResult(image: cropped, rect: rect, displayID: displayID)
    }
}

enum CaptureError: LocalizedError {
    case displayNotFound(CGDirectDisplayID)
    case cropFailed
    case permissionDenied
    var errorDescription: String? {
        switch self {
        case .displayNotFound(let id): "找不到显示器 \(id)"
        case .cropFailed: "截图裁剪失败"
        case .permissionDenied: "屏幕录制权限未授权"
        }
    }
}
```

注意：`SCContentFilter(display:excludingWindows:)` 与 `display.scale`、`display.frame` 需真机验证（对应规范 POC-01/02）；如 API 签名不匹配，按 SDK 实际签名调整并记录。

```swift
import Foundation
import CoreGraphics
import ImageIO
import UniformTypeIdentifiers
import SQLite3

struct ScreenshotRecord: Identifiable, Hashable {
    let id: UUID
    let filename: String
    let width: Int
    let height: Int
    let fileSize: Int
    let capturedAt: Date
    var appName: String { "LENSO" }  // MVP：来源应用暂记 LENSO
}

final class ScreenshotStore {
    static let shared = ScreenshotStore()
    private let db: OpaquePointer?
    private let baseDir: URL

    init() {
        baseDir = FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0]
            .appendingPathComponent("LENSO", isDirectory: true)
        try? FileManager.default.createDirectory(at: baseDir.appendingPathComponent("Original"), withIntermediateDirectories: true)
        try? FileManager.default.createDirectory(at: baseDir.appendingPathComponent("Thumbnails"), withIntermediateDirectories: true)
        db = Self.openDB(at: baseDir.appendingPathComponent("lenso.sqlite"))
        Self.migrate(db: db)
    }

    // 建表：screenshots(id TEXT PK, filename TEXT, width INT, height INT, file_size INT, captured_at REAL)
    private static func migrate(db: OpaquePointer?) { /* sqlite3_exec CREATE TABLE IF NOT EXISTS */ }

    /// 保存：写 PNG 文件 → 生成 400px 宽缩略图 → INSERT。返回记录。
    func save(image: CGImage, capturedAt: Date) throws -> ScreenshotRecord
    func list() throws -> [ScreenshotRecord]     // ORDER BY captured_at DESC
    func loadImage(_ record: ScreenshotRecord) -> CGImage?      // 读原图
    func loadThumbnail(_ record: ScreenshotRecord) -> CGImage?  // 读缩略图
    func delete(_ record: ScreenshotRecord) throws              // 删文件 + DELETE
}
```
